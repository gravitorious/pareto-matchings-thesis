	\begin{frame}
		\frametitle{Αντικείμενο και βασικές προσεγγίσεις}
		
		Η διπλωματική μελετά προβλήματα ταιριάσματος σε γράφους, τα οποία εμφανίζονται σε εφαρμογές ανάθεσης, όπως:
		
		\begin{itemize}
			\item δάσκαλοι σε σχολεία,
			\item applicants σε κατοικίες,
			\item φοιτητές και καθηγητές σε projects.
		\end{itemize}
		
		\vspace{0.2cm}
		
		Η μελέτη ακολουθεί δύο συμπληρωματικές προσεγγίσεις:
		
		\begin{itemize}
			\item τον σχεδιασμό και την ανάλυση συνδυαστικών αλγορίθμων,
			\item τη διατύπωση μέσω γραμμικού και ακέραιου προγραμματισμού.
		\end{itemize}
		
		Στο τελευταίο μέρος χρησιμοποιείται επίσης πολυεδρική και υπολογιστική ανάλυση.
	\end{frame}
	
	\begin{frame}
		\frametitle{Δομή της εργασίας}
		
		Η εργασία χωρίζεται σε δύο βασικά μέρη.
		
		\vspace{0.2cm}
		
		\begin{itemize}
			\item \textbf{Πρώτο μέρος:}
			μέγιστο ταίριασμα σε διμερείς γράφους και σταθερός γάμος, τόσο αλγοριθμικά όσο και μέσω γραμμικού προγραμματισμού.
			
			\vspace{0.2cm}
			
			\item \textbf{Δεύτερο μέρος:}
			Pareto βέλτιστα ταιριάσματα σε ένα μοντέλο ανάθεσης αδιαίρετων κατοικιών, στο οποίο μόνο οι applicants έχουν προτιμήσεις.
		\end{itemize}
		
		\begin{block}{Κύριοι στόχοι του δεύτερου μέρους}
			Μελετάμε τη γραμμική χαλάρωση, το separation problem και τον χώρο λύσεων των POM vectors για μικρά instances.
		\end{block}
	\end{frame}
	
	% Μέγιστο διμερές ταίριασμα
	
	\begin{frame}
		\frametitle{Μέγιστο ταίριασμα σε διμερείς γράφους}
		
		Έστω ένας διμερής γράφος
		\[
		G=(V_1\cup V_2,E).
		\]
		
		Ένα \textbf{ταίριασμα} είναι ένα σύνολο ακμών στο οποίο κάθε κορυφή συμμετέχει σε το πολύ μία ακμή.
		
		\vspace{0.2cm}
		
		Ένα ταίριασμα $M$ ονομάζεται:
		
		\begin{itemize}
			\item \textbf{μεγιστικό}, αν δεν μπορεί να προστεθεί άλλη ακμή,
			\item \textbf{μέγιστο}, αν κανένα άλλο ταίριασμα δεν έχει περισσότερες ακμές.
		\end{itemize}
		
		Κάθε μέγιστο ταίριασμα είναι μεγιστικό, αλλά το αντίστροφο δεν ισχύει πάντα.
	\end{frame}
	
	\begin{frame}
		\frametitle{Αυξητικά μονοπάτια και θεώρημα του Berge}
		
		Ένα μονοπάτι είναι \textbf{εναλλασσόμενο} ως προς το $M$, όταν οι ακμές του ανήκουν εναλλάξ στο $M$ και στο $E\setminus M$.
		
		\vspace{0.2cm}
		
		Είναι \textbf{αυξητικό} όταν τα δύο άκρα του είναι ελεύθερα και η πρώτη και η τελευταία ακμή του δεν ανήκουν στο $M$.
		
		Αν εναλλάξουμε τις ακμές του, τότε:
		\[
		|M|\longrightarrow |M|+1.
		\]
		
		\begin{block}{Θεώρημα του Berge}
			Ένα ταίριασμα $M$ είναι μέγιστο αν και μόνο αν δεν υπάρχει αυξητικό μονοπάτι ως προς το $M$.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Αλγόριθμος αυξητικών μονοπατιών}
		
		Ο βασικός αλγόριθμος ξεκινά από το κενό ταίριασμα και επαναλαμβάνει:
		
		\begin{enumerate}
			\item επιλέγει μία ελεύθερη κορυφή του $V_1$,
			\item αναζητά με DFS ένα αυξητικό μονοπάτι,
			\item αν βρεθεί, εναλλάσσει τις ακμές του.
		\end{enumerate}
		
		Όταν δεν υπάρχει αυξητικό μονοπάτι, το θεώρημα του Berge εγγυάται ότι το ταίριασμα είναι μέγιστο.
		
		\begin{block}{Πολυπλοκότητα}
			Η συνολική χρονική πολυπλοκότητα είναι
			\[
			O(nm),
			\]
			όπου $n$ είναι ο αριθμός των κορυφών και $m$ ο αριθμός των ακμών.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Αλγόριθμος Hopcroft--Karp}
		
		Ο Hopcroft--Karp βρίσκει πολλά αυξητικά μονοπάτια σε κάθε φάση.
		
		\vspace{0.3cm}
		
		Κάθε φάση περιλαμβάνει:
		
		\begin{enumerate}
			\item BFS για την κατασκευή ενός πολυεπίπεδου δικτύου που περιέχει τα συντομότερα αυξητικά μονοπάτια,
			\item DFS για την εύρεση ενός μεγιστικού συνόλου κορυφοξένων συντομότερων αυξητικών μονοπατιών.
		\end{enumerate}
		
		\vspace{0.3cm}
		
		Οι ακμές όλων των μονοπατιών εναλλάσσονται ταυτόχρονα.
	\end{frame}
	
	\begin{frame}
		\frametitle{Πολυεπίπεδο δίκτυο και πολυπλοκότητα}
		
		\begin{center}
			\includegraphics[
			width=0.68\textwidth,
			height=0.48\textheight,
			keepaspectratio
			]{pic1}
		\end{center}
		
		Κάθε φάση απαιτεί $O(n+m)$ χρόνο και ο αριθμός των φάσεων είναι $O(\sqrt{n})$.
		
		\begin{block}{Συνολική πολυπλοκότητα}
			\[
			O\bigl((n+m)\sqrt{n}\bigr).
			\]
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Σύνδεση με το πρόβλημα μέγιστης ροής}
		
		Το μέγιστο διμερές ταίριασμα ανάγεται σε πρόβλημα μέγιστης ροής.
		
		Κατασκευάζουμε ένα δίκτυο με:
		
		\begin{itemize}
			\item μία πηγή που συνδέεται με όλες τις κορυφές του $V_1$,
			\item τις ακμές του γράφου κατευθυνόμενες από το $V_1$ προς το $V_2$,
			\item μία καταβόθρα στην οποία συνδέονται όλες οι κορυφές του $V_2$,
			\item χωρητικότητα ίση με $1$ σε κάθε ακμή.
		\end{itemize}
		
		\begin{block}{Αποτέλεσμα}
			Η τιμή μιας μέγιστης ακέραιας ροής είναι ίση με την πληθικότητα ενός μέγιστου ταιριάσματος.
		\end{block}
	\end{frame}
	
	% Σταθερός γάμος
	
	\begin{frame}
		\frametitle{Σταθερός γάμος και blocking pairs}
		
		Δίνονται δύο σύνολα ίδιου μεγέθους:
		\[
		U=\{m_1,\ldots,m_n\},
		\qquad
		V=\{w_1,\ldots,w_n\},
		\]
		μαζί με αυστηρές σειρές προτίμησης.
		
		Ένας γάμος είναι ένα τέλειο ταίριασμα μεταξύ των δύο συνόλων.
		
		\vspace{0.2cm}
		
		Ένα ζεύγος $(m_i,w_j)\notin M$ είναι \textbf{blocking pair} όταν:
		
		\begin{itemize}
			\item ο $m_i$ προτιμά τη $w_j$ από τη σύντροφό του,
			\item η $w_j$ προτιμά τον $m_i$ από τον σύντροφό της.
		\end{itemize}
		
		Ένας γάμος είναι \textbf{σταθερός} όταν δεν έχει blocking pair. Ένας σταθερός γάμος δεν είναι απαραίτητα μοναδικός.
	\end{frame}
	
	\begin{frame}
		\frametitle{Ο αλγόριθμος Gale--Shapley}
		
		Στην εκδοχή όπου προτείνουν οι άνδρες:
		
		\begin{enumerate}
			\item ένας ελεύθερος άνδρας προτείνει στην καλύτερη γυναίκα στην οποία δεν έχει προτείνει ακόμη,
			\item αν η γυναίκα είναι ελεύθερη, αποδέχεται την πρόταση,
			\item αν είναι δεσμευμένη, κρατά τον άνδρα που προτιμά και απορρίπτει τον άλλο,
			\item ο άνδρας που απορρίπτεται γίνεται ξανά ελεύθερος.
		\end{enumerate}
		
		\vspace{0.3cm}
		
		Η διαδικασία επαναλαμβάνεται μέχρι να μην υπάρχει ελεύθερος άνδρας.
	\end{frame}
	
	\begin{frame}
		\frametitle{Βασικές ιδιότητες και ορθότητα}
		
		Κατά τη διάρκεια του Gale--Shapley:
		
		\begin{itemize}
			\item κάθε άνδρας προτείνει σε κάθε γυναίκα το πολύ μία φορά,
			\item κάθε γυναίκα αλλάζει προσωρινό σύντροφο μόνο για κάποιον που προτιμά περισσότερο,
			\item από τη στιγμή που μία γυναίκα δεχθεί πρόταση, δεν γίνεται ξανά ελεύθερη.
		\end{itemize}
		
		Επομένως, πραγματοποιούνται το πολύ
		\[
		n^2
		\]
		προτάσεις και ο αλγόριθμος τερματίζει.
		
		\begin{block}{Βασικό αποτέλεσμα}
			Ο Gale--Shapley τερματίζει πάντοτε και επιστρέφει έναν σταθερό γάμο.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Υλοποίηση και πολυπλοκότητα}
		
		Για την αποδοτική υλοποίηση χρησιμοποιούμε:
		
		\begin{itemize}
			\item μία ουρά με τους ελεύθερους άνδρες,
			\item έναν δείκτη $\mathit{ptr}[m]$ για την επόμενη πρόταση κάθε άνδρα,
			\item τους πίνακες $\mathit{manPartner}$ και $\mathit{womanPartner}$,
			\item έναν πίνακα $\mathit{rank}$ για συγκρίσεις σε χρόνο $O(1)$.
		\end{itemize}
		
		Η προεπεξεργασία απαιτεί $O(n^2)$ χρόνο, ενώ κάθε πρόταση εκτελείται σε χρόνο $O(1)$.
		
		\begin{block}{Συνολική πολυπλοκότητα}
			\[
			O(n^2).
			\]
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Man-optimality και female-pessimality}
		
		Στην εκδοχή όπου προτείνουν οι άνδρες:
		
		\begin{block}{Man-optimality}
			Κάθε άνδρας αντιστοιχίζεται με την καλύτερη γυναίκα που μπορεί να έχει σε οποιονδήποτε σταθερό γάμο.
		\end{block}
		
		\begin{block}{Female-pessimality}
			Κάθε γυναίκα αντιστοιχίζεται με τον λιγότερο προτιμώμενο άνδρα που μπορεί να έχει σε οποιονδήποτε σταθερό γάμο.
		\end{block}
		
		Επομένως, η πλευρά που πραγματοποιεί τις προτάσεις ευνοείται από τον αλγόριθμο.
	\end{frame}
	
	% Πολυεδρική προσέγγιση στα κλασικά προβλήματα
	
	\begin{frame}
		\frametitle{Πολυεδρική προσέγγιση}
		
		Τα προβλήματα ταιριάσματος μπορούν να μελετηθούν και μέσω γραμμικού και ακέραιου προγραμματισμού.
		
		\vspace{0.3cm}
		
		Η βασική διαδικασία είναι:
		
		\begin{enumerate}
			\item διατυπώνουμε το πρόβλημα ως ακέραιο πρόγραμμα,
			\item αφαιρούμε τους περιορισμούς ακεραιότητας,
			\item μελετάμε τη γραμμική χαλάρωση και τα ακραία σημεία της.
		\end{enumerate}
		
		\vspace{0.3cm}
		
		Αν η γραμμική χαλάρωση είναι ακέραια, τότε το αρχικό διακριτό πρόβλημα μπορεί να λυθεί μέσω γραμμικού προγραμματισμού.
	\end{frame}
	
	\begin{frame}
		\frametitle{Διατύπωση του διμερούς ταιριάσματος}
		
		Έστω ένας διμερής γράφος
		\[
		G=(V_1\cup V_2,E)
		\]
		και $x_e=1$ αν η ακμή $e$ ανήκει στο ταίριασμα.
		
		Το πρόβλημα μέγιστου ταιριάσματος διατυπώνεται ως:
		\[
		\begin{aligned}
			\max \quad & \sum_{e\in E}x_e\\
			\text{s.t.}\quad
			& \sum_{e\in\delta(v)}x_e\leq 1,
			&& \forall v\in V,\\
			& x_e\in\{0,1\},
			&& \forall e\in E.
		\end{aligned}
		\]
		
		Αν $A$ είναι ο πίνακας πρόσπτωσης κορυφών-ακμών, η γραμμική χαλάρωση είναι:
		\[
		\max\{\mathbf{1}^Tx:Ax\leq\mathbf{1},\ x\geq 0\}.
		\]
	\end{frame}
	
	\begin{frame}
		\en
		\frametitle{Total Unimodularity και ακεραιότητα}
		\gr
		
		\begin{block}{Βασικό αποτέλεσμα}
			Ο πίνακας πρόσπτωσης κορυφών-ακμών ενός διμερούς γράφου είναι \en totally unimodular\gr.
		\end{block}
		
		Επειδή ο πίνακας περιορισμών είναι \en totally unimodular \gr και το δεξί μέλος $\mathbf{1}$ είναι ακέραιο, κάθε ακραίο σημείο της γραμμικής χαλάρωσης είναι ακέραιο.
		
		\vspace{0.3cm}
		
		Επομένως:
		\[
		\text{βέλτιστη τιμή LP}
		=
		\text{βέλτιστη τιμή IP}.
		\]
		
		\begin{block}{Συνέπεια}
			Ένα μέγιστο διμερές ταίριασμα μπορεί να υπολογιστεί λύνοντας μόνο το γραμμικό πρόγραμμα.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Το θεώρημα του K\"onig}
		
		Το δυϊκό της γραμμικής χαλάρωσης είναι:
		\[
		\begin{aligned}
			\min \quad & \mathbf{1}^Ty\\
			\text{s.t.}\quad
			& A^Ty\geq \mathbf{1},\\
			& y\geq 0.
		\end{aligned}
		\]
		
		Πρόκειται για τη γραμμική χαλάρωση του προβλήματος ελάχιστης κάλυψης κορυφών.
		
		\begin{block}{Θεώρημα του K\"onig}
			Σε κάθε διμερή γράφο, η πληθικότητα ενός μέγιστου ταιριάσματος είναι ίση με την πληθικότητα μιας ελάχιστης κάλυψης κορυφών.
		\end{block}
		
		Το αποτέλεσμα προκύπτει από την \en total unimodularity \gr και την ισχυρή δυϊκότητα.
	\end{frame}
	
	\begin{frame}
		\frametitle{Ακέραια διατύπωση του σταθερού γάμου}
		
		Για κάθε ζεύγος $(m,w)$, θέτουμε $x_{mw}=1$ αν ο $m$ αντιστοιχίζεται με τη $w$.
		
		Οι περιορισμοί ανάθεσης είναι:
		\[
		\sum_{w\in V}x_{mw}=1
		\quad \forall m\in U,
		\qquad
		\sum_{m\in U}x_{mw}=1
		\quad \forall w\in V.
		\]
		
		Για κάθε ζεύγος $(m,w)$ επιβάλλεται επίσης:
		\small
		\[
		x_{mw}
		+
		\sum_{\substack{w'\in V\\w'\succ_m w}}x_{mw'}
		+
		\sum_{\substack{m'\in U\\m'\succ_w m}}x_{m'w}
		\geq 1.
		\]
		\normalsize
		
		Ο περιορισμός απαιτεί είτε οι $m,w$ να είναι μαζί, είτε τουλάχιστον ένας από τους δύο να έχει αντιστοιχιστεί με προτιμότερο σύντροφο. Άρα το $(m,w)$ δεν μπορεί να είναι blocking pair.
	\end{frame}
	
	\begin{frame}
		\frametitle{Ακεραιότητα του stable matching polytope}
		
		Ορίζουμε:
		\[
		P_{SM}
		=
		\operatorname{conv}
		\left\{
		\chi^M:
		M \text{ είναι ένας σταθερός γάμος}
		\right\}.
		\]
		
		Έστω $Q_{SM}$ η γραμμική χαλάρωση της προηγούμενης διατύπωσης.
		
		\begin{block}{Θεώρημα Vande Vate--Rothblum}
			Για πλήρεις και αυστηρές λίστες προτιμήσεων ισχύει:
			\[
			Q_{SM}=P_{SM}.
			\]
		\end{block}
		
		Επομένως, κάθε ακραίο σημείο του $Q_{SM}$ είναι το διάνυσμα πρόσπτωσης ενός σταθερού γάμου.
	\end{frame}
	
	\begin{frame}
		\frametitle{Βελτιστοποίηση πάνω στους σταθερούς γάμους}
		
		Αν κάθε ζεύγος $(m,w)$ έχει βάρος $c_{mw}$, μπορούμε να λύσουμε:
		\[
		\begin{aligned}
			\max \quad &
			\sum_{m\in U}\sum_{w\in V}c_{mw}x_{mw}\\
			\text{s.t.}\quad &
			x\in Q_{SM}.
		\end{aligned}
		\]
		
		Επειδή το $Q_{SM}$ είναι ακέραιο, υπάρχει βέλτιστη λύση που αντιστοιχεί σε σταθερό γάμο.
		
		\begin{block}{Συμπέρασμα}
			Οι φυσικές γραμμικές χαλαρώσεις του διμερούς ταιριάσματος και του σταθερού γάμου είναι ακέραιες. Η εικόνα αυτή αλλάζει στα Pareto βέλτιστα ταιριάσματα.
		\end{block}
	\end{frame}
	
	% Pareto βέλτιστα ταιριάσματα
	
	\begin{frame}
		\frametitle{Μοντέλο ανάθεσης και Pareto βελτιστότητα}
		
		Δίνονται αιτούντες $A=\{a_1,\ldots,a_n\}$, κατοικίες $H$ και μία αυστηρή λίστα αποδεκτών κατοικιών $P(a)$ για κάθε $a\in A$.
		
		Μόνο οι αιτούντες έχουν προτιμήσεις, δεν απαιτείται $|A|=|H|$ και επιτρέπονται μη αντιστοιχισμένοι αιτούντες ή κατοικίες.
		
		\vspace{0.2cm}
		
		Ένα ταίριασμα $M'$ είναι \textbf{Pareto βελτίωση} του $M$ όταν:
		
		\begin{itemize}
			\item κανένας αιτών δεν βρίσκεται σε χειρότερη θέση στο $M'$,
			\item τουλάχιστον ένας αιτών βρίσκεται σε αυστηρά καλύτερη θέση.
		\end{itemize}
		
		Η μη αντιστοίχιση θεωρείται χειρότερη από κάθε αποδεκτή κατοικία. Ένα ταίριασμα είναι \textbf{Pareto βέλτιστο} όταν δεν επιδέχεται Pareto βελτίωση.
	\end{frame}
	
	\begin{frame}
		\frametitle{Χαρακτηρισμός της Pareto βελτιστότητας}
		
		\begin{block}{Χαρακτηρισμός}
			Ένα ταίριασμα είναι Pareto βέλτιστο αν και μόνο αν είναι:
			
			\begin{itemize}
				\item μεγιστικό,
				\item trade-in free,
				\item coalition free.
			\end{itemize}
		\end{block}
		
		\vspace{0.2cm}
		
		\begin{itemize}
			\item \textbf{Μεγιστικό:} δεν υπάρχει ελεύθερος αιτών και ελεύθερη αποδεκτή κατοικία.
			\item \textbf{Trade-in free:} κανένας αιτών δεν μπορεί να μετακινηθεί σε καλύτερη ελεύθερη κατοικία.
			\item \textbf{Coalition free:} δεν υπάρχει κυκλική ανταλλαγή που βελτιώνει αυστηρά όλους τους συμμετέχοντες.
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Plausible coalitions}
		
		Μία plausible coalition είναι ένα σύνολο ζευγών
		\[
		C=\{(a_{i_0},h_{i_0}),\ldots,(a_{i_{k-1}},h_{i_{k-1}})\},
		\]
		τέτοιο ώστε
		\[
		h_{i_{c+1}}>_{a_{i_c}}h_{i_c},
		\qquad c=0,\ldots,k-1,
		\]
		με κυκλική αρίθμηση των δεικτών.
		
		\vspace{0.3cm}
		
		Αν όλα τα ζεύγη του $C$ εμφανίζονται στο ίδιο ταίριασμα, οι συμμετέχοντες μπορούν να ανταλλάξουν κυκλικά τις κατοικίες τους και όλοι να βελτιωθούν.
		
		Με $\mathcal C$ συμβολίζουμε την οικογένεια όλων των plausible coalitions.
	\end{frame}
	
	\begin{frame}
		\frametitle{Ακέραια διατύπωση: matching και maximality}
		
		Για κάθε αποδεκτό ζεύγος $(a,h)$, η μεταβλητή $x_{ah}$ είναι ίση με $1$ αν το ζεύγος ανήκει στο ταίριασμα και με $0$ διαφορετικά.
		
		Ορίζουμε
		\[
		Q(h)=\{a\in A:h\in P(a)\}.
		\]
		
		Οι matching constraints είναι:
		\[
		\sum_{h\in P(a)}x_{ah}\leq 1,
		\qquad
		\sum_{a\in Q(h)}x_{ah}\leq 1.
		\]
		
		Για κάθε αποδεκτό ζεύγος $(a,h)$, η μεγιστικότητα επιβάλλεται από:
		\[
		\sum_{h'\in P(a)}x_{ah'}
		+
		\sum_{a'\in Q(h)\setminus\{a\}}x_{a'h}
		\geq 1.
		\]
		
		Έτσι, ο $a$ και η $h$ δεν μπορούν να είναι ταυτόχρονα ελεύθεροι.
	\end{frame}
	
	\begin{frame}
		\frametitle{Trade-in και coalition constraints}
		
		Η trade-in free ιδιότητα εκφράζεται από:
		\[
		\sum_{a'\in Q(h)}x_{a'h}
		-
		\sum_{\substack{h'\in P(a)\\h>_a h'}}x_{ah'}
		\geq 0.
		\]
		
		Για κάθε plausible coalition $C\in\mathcal C$ επιβάλλουμε:
		\[
		\sum_{(a,h)\in C}x_{ah}\leq |C|-1.
		\]
		
		Άρα τουλάχιστον ένα ζεύγος κάθε plausible coalition απουσιάζει από το ταίριασμα.
		
		\begin{block}{Ορθότητα της διατύπωσης}
			Τα εφικτά ακέραια σημεία της διατύπωσης είναι ακριβώς τα διανύσματα πρόσπτωσης των Pareto βέλτιστων ταιριασμάτων, τα οποία ονομάζουμε POM vectors.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Μη ακεραιότητα της γραμμικής χαλάρωσης}
		
		Η γραμμική χαλάρωση προκύπτει αντικαθιστώντας
		\[
		x_{ah}\in\{0,1\}
		\qquad\text{με}\qquad
		x_{ah}\geq 0.
		\]
		
		Οι matching constraints συνεπάγονται ήδη ότι
		\[
		x_{ah}\leq 1.
		\]
		
		\vspace{0.3cm}
		
		Σε αντίθεση με το maximum bipartite matching και το stable marriage problem, η συγκεκριμένη χαλάρωση δεν είναι γενικά ακέραια.
		
		\begin{block}{Αποτέλεσμα}
			Υπάρχουν instances για τα οποία η χαλάρωση έχει κλασματικά ακραία σημεία που δεν ανήκουν στο $P_{\mathrm{POM}}$.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Ένα κλασματικό ακραίο σημείο}
		
		Για ένα instance με $5$ αιτούντες και $5$ κατοικίες, προέκυψε το κλασματικό ακραίο σημείο:
		
		\small
		\[
		\begin{aligned}
			x_{14}&=1, &
			x_{22}&=\frac{2}{3}, &
			x_{33}&=\frac{1}{3}, &
			x_{31}&=\frac{2}{3},\\
			x_{43}&=\frac{1}{3}, &
			x_{45}&=\frac{2}{3}, &
			x_{52}&=\frac{1}{3}, &
			x_{51}&=\frac{1}{3},
		\end{aligned}
		\]
		\normalsize
		
		με όλες τις υπόλοιπες μεταβλητές ίσες με $0$.
		
		\vspace{0.3cm}
		
		Το σημείο ικανοποιεί τη βασική γραμμική χαλάρωση, αλλά δεν μπορεί να εκφραστεί ως κυρτός συνδυασμός POM vectors.
	\end{frame}
	
	\begin{frame}
		\frametitle{Παραγοντικό πλήθος plausible coalitions}
		
		Θεωρούμε ένα instance με πέντε αιτούντες και είκοσι πέντε κατοικίες. Οι κατοικίες χωρίζονται σε πέντε blocks:
		
		\small
		\[
		\begin{aligned}
			H_1&=\{1,2,3,4,5\},
			& H_2&=\{6,7,8,9,10\},\\
			H_3&=\{11,12,13,14,15\},
			& H_4&=\{16,17,18,19,20\},\\
			H_5&=\{21,22,23,24,25\}.
		\end{aligned}
		\]
		\normalsize
		
		Με $h_{i,j}$ συμβολίζουμε την $j$-οστή κατοικία του block $H_i$.
		
		Υποθέτουμε ότι ο αιτών $a_i$ προτιμά όλες τις κατοικίες του $H_{i+1}$ από όλες τις κατοικίες του $H_i$, όπου οι δείκτες λαμβάνονται κυκλικά ($H_6=H_1$).
	\end{frame}
	
	\begin{frame}
		\frametitle{Κατασκευή των plausible coalitions}
		
		Για κάθε μετάθεση $\pi$ του συνόλου $\{1,2,3,4,5\}$, ορίζουμε:
		
		\small
		\[
		\begin{aligned}
			C_\pi=\{&(a_1,h_{1,\pi(1)}),(a_2,h_{2,\pi(2)}),
			(a_3,h_{3,\pi(3)}),\\
			&(a_4,h_{4,\pi(4)}),(a_5,h_{5,\pi(5)})\}.
		\end{aligned}
		\]
		\normalsize
		
		Ο $a_1$ προτιμά την $h_{2,\pi(2)}$ από την $h_{1,\pi(1)}$, ο $a_2$ προτιμά την $h_{3,\pi(3)}$ από την $h_{2,\pi(2)}$ και, κυκλικά, ο $a_5$ προτιμά την $h_{1,\pi(1)}$ από την $h_{5,\pi(5)}$.
		
		Άρα κάθε μετάθεση ορίζει μία διαφορετική plausible coalition.
		
		\begin{block}{Πλήθος coalitions}
			Η κατασκευή δίνει τουλάχιστον
			\[
			5!=120
			\]
			διαφορετικές plausible coalitions. Με $n$ αιτούντες και $n$ κατοικίες σε κάθε block δίνει τουλάχιστον $n!$.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Separation των coalition constraints}
		
		Η οικογένεια $\mathcal C$ μπορεί να έχει παραγοντικό μέγεθος. Για ένα δοσμένο σημείο $x^*$ αναζητούμε έναν παραβιασμένο coalition constraint.
		
		Κατασκευάζουμε έναν βοηθητικό κατευθυνόμενο γράφο με:
		\[
		V_G=\{(a,h):h\in P(a)\}.
		\]
		
		Προσθέτουμε την ακμή
		\[
		(a,h)\longrightarrow(a',h')
		\]
		όταν $a\neq a'$ και
		\[
		h'>_a h.
		\]
		
		Κάθε plausible coalition αντιστοιχεί σε έναν κατευθυνόμενο κύκλο. Οι επιπλέον κύκλοι με επαναλαμβανόμενους αιτούντες ή κατοικίες δεν δίνουν παραβιασμένο cut σε σημείο που ικανοποιεί τους matching constraints.
	\end{frame}
	
	\begin{frame}
		\frametitle{Ελάχιστος κύκλος και αλγόριθμος διαχωρισμού}
		
		Για το σημείο $x^*$ θέτουμε:
		\[
		w(a,h)=1-x^*_{ah}.
		\]
		
		Ένας coalition constraint παραβιάζεται αν και μόνο αν:
		\[
		\sum_{(a,h)\in C}(1-x^*_{ah})<1.
		\]
		
		Άρα αναζητούμε κατευθυνόμενο κύκλο ελάχιστου βάρους. Θέτουμε:
		\[
		w^*(u,v)=w(v).
		\]
		
		\begin{enumerate}
			\item Εκτελούμε τον αλγόριθμο Floyd--Warshall.
			\item Για κάθε ακμή $(u,v)$ εξετάζουμε την τιμή $w^*(u,v)+D_{v,u}$.
			\item Αν η ελάχιστη τιμή είναι μικρότερη από $1$, ανακτούμε την παραβιασμένη coalition.
		\end{enumerate}
	\end{frame}
	
	\begin{frame}
		\frametitle{Πολυπλοκότητα του separation}
		
		Έστω
		\[
		n=|A|,
		\qquad
		L=\max_{a\in A}|P(a)|.
		\]
		
		Ο βοηθητικός γράφος έχει το πολύ $nL$ κορυφές και
		\[
		O(n^2L^2)
		\]
		ακμές.
		
		Η κυρίαρχη πράξη είναι η εκτέλεση του Floyd--Warshall.
		
		\begin{block}{Αποτέλεσμα}
			Το separation problem για τους coalition constraints επιλύεται σε χρόνο
			\[
			O(n^3L^3).
			\]
		\end{block}
		
		Για $n$ αιτούντες και $n$ κατοικίες, η χειρότερη περίπτωση είναι $O(n^6)$.
	\end{frame}
	
	\begin{frame}
		\frametitle{Το πολύτοπο και η παραγωγή των POM vectors}
		
		Για ένα σταθερό instance ορίζουμε:
		\[
		P_{\mathrm{POM}}
		=
		\operatorname{conv}
		\left\{
		x^M:
		M\text{ είναι Pareto βέλτιστο ταίριασμα}
		\right\}.
		\]
		
		Τα POM vectors παράγονται μέσω του Serial Dictatorship:
		
		\begin{enumerate}
			\item επιλέγουμε μία διάταξη των αιτούντων,
			\item κάθε αιτών επιλέγει την καλύτερη διαθέσιμη αποδεκτή κατοικία,
			\item αν δεν υπάρχει τέτοια κατοικία, παραμένει χωρίς αντιστοίχιση.
		\end{enumerate}
		
		\begin{block}{Χαρακτηρισμός}
			Κάθε αποτέλεσμα του Serial Dictatorship είναι Pareto βέλτιστο και κάθε Pareto βέλτιστο ταίριασμα προκύπτει από κάποια διάταξη.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Υπολογιστική πολυεδρική ανάλυση}
		
		Για κάθε μικρό instance ακολουθούμε την εξής διαδικασία:
		
		\begin{enumerate}
			\item εκτελούμε το Serial Dictatorship για όλες τις διατάξεις,
			\item αφαιρούμε τα διπλότυπα POM vectors,
			\item κατασκευάζουμε των χώρο λύσεων τους (πολύτοπο) με το SageMath,
			\item εξάγουμε την $H$-description του $P_{\mathrm{POM}}$,
			\item συγκρίνουμε την περιγραφή με κλασματικά σημεία της χαλάρωσης.
		\end{enumerate}
		
		\begin{block}{Στόχος}
			Εντοπίζουμε affine equalities και έγκυρες ανισώσεις που δεν επιβάλλονται από την αρχική χαλάρωση.
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{First-choice και prefix-cover inequalities}
		
		Αν μία κατοικία $h$ είναι πρώτη επιλογή κάποιου αιτούντος, τότε σε κάθε POM vector:
		\[
		\sum_{b\in Q(h)}x_{bh}=1.
		\]
		
		Για $a\in A$ και $h\in P(a)$ ορίζουμε:
		\[
		B_a(h)=\{g\in P(a):g>_a h\}.
		\]
		
		Η προηγούμενη ιδιότητα γενικεύεται στην έγκυρη ανίσωση:
		\[
		\sum_{b\in Q(h)}x_{bh}
		+
		\sum_{g\in B_a(h)}x_{ag}
		\geq 1.
		\]
		
		Οι ανισώσεις αυτές ονομάζονται \textbf{prefix-cover inequalities}.
	\end{frame}
	
	\begin{frame}
		\frametitle{Άνω φράγμα για τη διάσταση}
		
		Έστω $F$ το σύνολο των κατοικιών που εμφανίζονται ως πρώτη επιλογή τουλάχιστον ενός αιτούντος.
		
		Για κάθε $h\in F$ ισχύει η affine equality:
		\[
		\sum_{a\in Q(h)}x_{ah}=1.
		\]
		
		Οι εξισώσεις αυτές είναι γραμμικά ανεξάρτητες.
		
		\begin{block}{Αποτέλεσμα}
			\[
			\dim(P_{\mathrm{POM}})
			\leq
			|E|-|F|.
			\]
		\end{block}
		
		Σε αρκετά πειραματικά instances η πραγματική διάσταση ήταν ακόμη μικρότερη.
	\end{frame}
	
	\begin{frame}
		\frametitle{Safe-holder inequalities}
		
		Έστω ότι ο αιτών $a$ λαμβάνει την κατοικία $h$ και προτιμά την κατοικία $g$ από την $h$.
		
		Ορίζουμε:
		\[
		S(a,h,g)
		=
		\{b\in Q(g)\setminus\{a\}:
		h\notin P(b)\ \text{ή}\ g>_b h\}.
		\]
		
		Τότε η ακόλουθη ανίσωση είναι έγκυρη για το $P_{\mathrm{POM}}$:
		\[
		x_{ah}
		\leq
		\sum_{b\in S(a,h,g)}x_{bg}.
		\]
		
		\begin{block}{Ερμηνεία}
			Αν ο $a$ λαμβάνει την $h$, τότε η καλύτερη κατοικία $g$ πρέπει να έχει ανατεθεί σε safe holder, ώστε να μην προκύπτει trade-in ή coalition δύο αιτούντων.
		\end{block}
	\end{frame}
	
	% Facet-defining inequalities for a cyclic family
	
	\begin{frame}
		\frametitle{Η κυκλική οικογένεια $\mathcal{I}_n$}
		
		Για κάθε $n\geq 3$, θεωρούμε:
		\[
		H=\{h_1,\ldots,h_n\},
		\qquad
		A=\{a_1,\ldots,a_n,b_1,\ldots,b_n\},
		\]
		με κυκλικούς δείκτες και προτιμήσεις:
		\[
		P(a_i)=\langle h_{i+1},h_i\rangle,
		\qquad
		P(b_i)=\langle h_i\rangle.
		\]
		
		Χρησιμοποιούμε τη σημειογραφία:
		\[
		s_i=x_{a_i h_i},
		\qquad
		t_i=x_{a_i h_{i+1}},
		\qquad
		r_i=x_{b_i h_i}.
		\]
		
		Κάθε κατοικία ανατίθεται σε κάθε POM vector. Επομένως:
		\[
		s_i+t_{i-1}+r_i=1,
		\qquad i=1,\ldots,n,
		\]
		όπου $t_0=t_n$.
	\end{frame}
	
	\begin{frame}
		\frametitle{Το auxiliary graph $G_n$}
		
		Ορίζουμε το μη κατευθυνόμενο γράφημα:
		\[
		V(G_n)=\{s_1,\ldots,s_n,t_1,\ldots,t_n\},
		\]
		\[
		E(G_n)
		=
		\bigl\{\{s_i,t_i\}:i=1,\ldots,n\bigr\}
		\cup
		\bigl\{\{t_i,s_{i+1}\}:i=1,\ldots,n\bigr\}.
		\]
		
		Οι πρώτες ακμές εκφράζουν συγκρούσεις ως προς τους αιτούντες και οι δεύτερες συγκρούσεις ως προς τις κατοικίες.
		
		Το $G_n$ είναι ο μη κατευθυνόμενος κύκλος:
		\[
		s_1-t_1-s_2-t_2-\cdots-s_n-t_n-s_1.
		\]
		
		Επομένως, οι μεταβλητές $s_i,t_i$ που λαμβάνουν την τιμή $1$ πρέπει να σχηματίζουν independent set του $G_n$.
	\end{frame}
	
	\begin{frame}
		\frametitle{Ακριβής χαρακτηρισμός των POM vectors}
		
		Για $x\in\{0,1\}^{3n}$, θέτουμε:
		\[
		S(x)=\{s_i:s_i=1\}\cup\{t_i:t_i=1\}.
		\]
		
		\begin{block}{Χαρακτηρισμός}
			Το $x$ είναι POM vector της $\mathcal I_n$ αν και μόνο αν:
			
			\begin{enumerate}
				\item το $S(x)$ είναι independent set του $G_n$,
				\item $S(x)\neq\{s_1,\ldots,s_n\}$,
				\item $r_i=1-s_i-t_{i-1}$ για κάθε $i=1,\ldots,n$.
			\end{enumerate}
		\end{block}
		
		Το εξαιρούμενο σύνολο αντιστοιχεί στην κυκλική coalition των αιτούντων $a_1,\ldots,a_n$.
	\end{frame}
	
	\begin{frame}
		\frametitle{Διάσταση του πολυτόπου}
		
		Το instance $\mathcal I_n$ έχει $3n$ αποδεκτά ζεύγη.
		
		Οι $n$ γραμμικά ανεξάρτητες affine equalities
		\[
		s_i+t_{i-1}+r_i=1,
		\qquad i=1,\ldots,n,
		\]
		δίνουν:
		\[
		\dim\bigl(P_{\mathrm{POM}}(\mathcal I_n)\bigr)\leq 2n.
		\]
		
		Από την άλλη πλευρά, μπορούν να κατασκευαστούν $2n+1$ affinely independent POM vectors.
		
		\begin{block}{Αποτέλεσμα}
			Για κάθε $n\geq 3$,
			\[
			\dim\bigl(P_{\mathrm{POM}}(\mathcal I_n)\bigr)=2n.
			\]
		\end{block}
	\end{frame}
	
	\begin{frame}
		\frametitle{Cycle-break inequalities}
		
		Για κάθε $j\in\{1,\ldots,n\}$, θεωρούμε την ανίσωση:
		\[
		t_j+\sum_{i=1}^{n}r_i\geq 1.
		\]
		
		\begin{block}{Ερμηνεία}
			Σε κάθε Pareto βέλτιστο ταίριασμα:
			
			\begin{itemize}
				\item είτε ο $a_j$ λαμβάνει την πρώτη επιλογή $h_{j+1}$,
				\item είτε τουλάχιστον ένας $b_i$ λαμβάνει την κατοικία $h_i$.
			\end{itemize}
		\end{block}
		
		Αν $t_j=0$ και $r_i=0$ για κάθε $i$, οι εξισώσεις των κατοικιών επιβάλλουν διαδοχικά:
		\[
		s_1=\cdots=s_n=1.
		\]
		
		Τότε οι αιτούντες $a_1,\ldots,a_n$ σχηματίζουν κυκλική coalition.
	\end{frame}
	
	\begin{frame}
		\frametitle{Facet-defining αποτέλεσμα}
		
		\begin{block}{Facet theorem}
			Για κάθε $n\geq 3$ και κάθε $j\in\{1,\ldots,n\}$, η cycle-break inequality
			\[
			t_j+\sum_{i=1}^{n}r_i\geq 1
			\]
			ορίζει facet του $P_{\mathrm{POM}}(\mathcal I_n)$.
		\end{block}
		
		Στο affine hull του πολυτόπου, η ανίσωση είναι ισοδύναμη με:
		\[
		\sum_{i=1}^{n}s_i
		+
		\sum_{\substack{i=1\\i\neq j}}^{n}t_i
		\leq n-1.
		\]
		
		Επειδή $\dim(P_{\mathrm{POM}}(\mathcal I_n))=2n$, η face που ορίζει έχει διάσταση $2n-1$.
	\end{frame}
	
	\begin{frame}
		\frametitle{Αποκοπή ενός κλασματικού σημείου}
		
		Θεωρούμε το κλασματικό σημείο $\bar{x}$ με:
		\[
		\bar{s}_i=\frac{1}{2},
		\qquad
		\bar{t}_i=\frac{1}{2},
		\qquad
		\bar{r}_i=0,
		\qquad i=1,\ldots,n.
		\]
		
		Το σημείο αυτό ικανοποιεί:
		
		\begin{itemize}
			\item τους matching, maximality, trade-in και coalition constraints,
			\item όλες τις prefix-cover inequalities,
			\item όλες τις safe-holder inequalities.
		\end{itemize}
		
		Ωστόσο, για κάθε $j=1,\ldots,n$:
		\[
		\bar{t}_j+\sum_{i=1}^{n}\bar{r}_i
		=
		\frac{1}{2}
		<1.
		\]
		
		Επομένως, οι cycle-break inequalities ενισχύουν αυστηρά την προηγούμενη γραμμική χαλάρωση.
	\end{frame}
	
	% Συμπεράσματα
	
	\begin{frame}
		\frametitle{Συνολικά συμπεράσματα}
		
		\begin{itemize}
			\item Οι Hopcroft--Karp και Gale--Shapley δίνουν αποδοτικούς συνδυαστικούς αλγορίθμους για δύο κλασικά προβλήματα.
			\item Οι φυσικές LP χαλαρώσεις του bipartite matching και του stable marriage είναι ακέραιες, ενώ η χαλάρωση των POMs δεν είναι γενικά ακέραια.
			\item Το separation των coalition constraints επιλύεται σε χρόνο $O(n^3L^3)$.
			\item Η υπολογιστική ανάλυση οδήγησε στο φράγμα διάστασης και στις prefix-cover και safe-holder inequalities.
			\item Στην κυκλική οικογένεια, το πολύτοπο έχει διάσταση $2n$ και οι cycle-break inequalities είναι facet-defining.
		\end{itemize}
		
	\end{frame}
	
	\begin{frame}
		\frametitle{Μελλοντική έρευνα}
		
		Βασικές κατευθύνσεις για περαιτέρω μελέτη είναι:
		
		\begin{itemize}
			\item ο χαρακτηρισμός των συνθηκών υπό τις οποίες οι prefix-cover ή safe-holder inequalities ορίζουν facets,
			\item η επέκταση των cycle-break inequalities σε γενικότερες γραφοθεωρητικές δομές,
			\item ο ακριβέστερος χαρακτηρισμός του affine hull, της διάστασης και νέων facets του $P_{\mathrm{POM}}$,
			\item η υπολογιστική αξιολόγηση των νέων inequalities σε μεγαλύτερα instances.
		\end{itemize}
		
		\begin{block}{Υπολογιστική βάση}
			Το υπάρχον framework παράγει τα POM vectors μικρών instances, κατασκευάζει την περιοχή εφικτών λύσεων που είναι ένα πολύτοπο και επιτρέπει τον έλεγχο νέων πολυεδρικών conjectures.
		\end{block}
	\end{frame}
